\begin{frame}[fragile]\frametitle{Commands 1: Setup and Memory}
\footnotesize
\begin{center}
\adjustbox{max width=\linewidth}{%
\begin{tabular}{|l|p{0.40\linewidth}|p{0.28\linewidth}|}
\hline
\textbf{Command} & \textbf{What it does} & \textbf{When to use} \\
\hline
\texttt{/doctor} & Checks the installation; inside a project it also tidies \texttt{CLAUDE.md} (duplicates, items that belong in skills) & Once per project \\
\hline
\texttt{@path/file} in \texttt{CLAUDE.md} & Imports another file instead of copying its text & Avoid duplicated rules \\
\hline
\texttt{CLAUDE.local.md} & Your own preferences, kept out of git & Habits the team did not agree to \\
\hline
\texttt{\textasciitilde/.claude/CLAUDE.md} & Applies to every session in every project & Language, style preferences \\
\hline
\texttt{/memory} & Lists what Claude remembered; promote good items to \texttt{CLAUDE.md} & Audit now and then \\
\hline
\texttt{/statusline} & Builds a status line: project, branch, model, context used & Set once, keep forever \\
\hline
\end{tabular}
}
\end{center}
Keep \texttt{CLAUDE.md} lean: ask of each line, ``needed for every task? could Claude find it alone?''
\end{frame}

\begin{frame}[fragile]\frametitle{Commands 2: Planning and Steering}
\footnotesize
\begin{center}
\adjustbox{max width=\linewidth}{%
\begin{tabular}{|l|p{0.40\linewidth}|p{0.28\linewidth}|}
\hline
\textbf{Command} & \textbf{What it does} & \textbf{When to use} \\
\hline
\texttt{Ctrl+G} & Opens the plan as Markdown in your editor; edit, save, Claude carries on & Fix one line, not a chat \\
\hline
``Interview me'' & Claude asks questions with its ask-user tool until no assumption is hidden & Before a big plan \\
\hline
\texttt{/rewind}, \texttt{Esc Esc} & Goes back to an earlier prompt and drops the later code from context & Wrong direction taken \\
\hline
\texttt{/model} + effort & Chooses model and reasoning effort for the thread & Plan big, implement cheaper \\
\hline
\texttt{ultrathink} & Maximum effort for that one prompt only & One hard bug \\
\hline
\texttt{Ctrl+S} & Stashes the prompt you are typing; press again to bring it back & Claude asks you mid-typing \\
\hline
\texttt{/btw} & Side question that does not wait or fill the context & While a task runs \\
\hline
\end{tabular}
}
\end{center}
\end{frame}

\begin{frame}[fragile]\frametitle{Commands 3: Context, Cost, Shell}
\footnotesize
\begin{center}
\adjustbox{max width=\linewidth}{%
\begin{tabular}{|l|p{0.40\linewidth}|p{0.28\linewidth}|}
\hline
\textbf{Command} & \textbf{What it does} & \textbf{When to use} \\
\hline
\texttt{/context} & Shows what fills the window; run it in an empty session & Spot unused skills and MCPs \\
\hline
\texttt{/compact} & Summarises history on your terms & After each finished part \\
\hline
\texttt{/status} & Usage and cost by model, subagents included & End of a session \\
\hline
\texttt{!} prefix & Runs a shell command directly, no reasoning; output enters the context & \texttt{!npm test}, \texttt{!ls} \\
\hline
\texttt{/diff} & Side pane of the files Claude changed & Review as it edits \\
\hline
\end{tabular}
}
\end{center}
Context quality is said to fall as the window fills (a rule of thumb around 40\%, an opinion, not a measured limit): compact early.
\end{frame}
