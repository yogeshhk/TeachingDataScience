\begin{frame}[fragile]\frametitle{While Writing: You Write}
\begin{itemize}
\item The sentences, the examples, the structure of each chapter: all yours.
\item Why hold this line:
\begin{itemize}
    \item Your voice is what a reader buys.
    \item You must stand behind every claim, and you can only do that for words you wrote.
    \item Publishers and journals have rules on AI-written text; many restrict it or require disclosure.
\end{itemize}
\end{itemize}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[fragile]\frametitle{What AI May Do While You Write}
\begin{columns}
    \begin{column}[T]{0.48\linewidth}
    \textbf{Allowed}
    \begin{itemize}
    \item Web searches for a fact, a date or a source to open and read
    \item Fact checks of claims you wrote, against sources you open
    \item Spelling and grammar checks that list the errors
    \item Preparing diagrams and figures
    \end{itemize}
    \end{column}
    \begin{column}[T]{0.48\linewidth}
    \textbf{Not allowed}
    \begin{itemize}
    \item Drafting paragraphs, examples or chapters
    \item Rewriting your sentences to ``improve'' them
    \end{itemize}
    \end{column}
\end{columns}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[fragile]\frametitle{CRAFT Example: Diagram Preparation}
\begin{itemize}
\item You decide what the figure must show. AI produces a first drawing you correct.
\item Check labels and numbers against your text.
\end{itemize}
\begin{lstlisting}[language=HTML]
You are a LaTeX TikZ expert (R). My book is printed
in black and white at 6 by 9 inches (C).

Draw a flowchart of four boxes: Agenda, Discuss,
Decide, Follow up, with arrows between them (A).
Example style: simple boxes, one-word labels (F).
Output only the TikZ code; no colours (T).
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]\frametitle{Quick Check: Allowed or Not?}
\begin{itemize}
\item Which of these is fine while you write?
\item[A.] ``Write the opening paragraph of Chapter 3 for me.''
\item[B.] ``Find the year this survey was first published, with a link.''
\item[C.] ``Make my paragraph sound smoother.''
\item[D.] ``Draw a two-by-two grid with these four labels.''
\end{itemize}
\end{frame}

\item \textbf{B and D are fine.} A search for a fact (then open the link and confirm) and a diagram are two of the permitted uses, with fact checks and grammar lists.
\item \textbf{A and C are not.} Both put someone else's words in your text, and in C they also change your voice.
\item If you are unsure, ask: ``Did the AI produce words that will appear in my text?'' If yes, do not.
\end{itemize}
\end{frame}

\begin{frame}[fragile]\frametitle{Quick Check: Analytical or Generative?}
\begin{itemize}
\item Which two requests only read and report?
\item[A.] ``List every place where a character's age conflicts.''
\item[B.] ``Write a stronger ending for Chapter 5.''
\item[C.] ``Check spelling and grammar and list the errors.''
\item[D.] ``Rewrite my introduction in a warmer tone.''
\end{itemize}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[fragile]\frametitle{Quick Check: Analytical or Generative?}
\begin{itemize}
\item \textbf{A and C are analytical.} They find problems and list them. You fix each one yourself.
\item \textbf{B and D are generative.} They put new words in your text.
\item Watch the verb: ``list the errors'' is analytical, ``correct my text'' is generative.
\end{itemize}
\end{frame}
