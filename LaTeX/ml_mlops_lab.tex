\begin{frame}[fragile]\frametitle{Your Turn: Deploy Your Own Model}
\begin{itemize}
\item Folder: \texttt{Code/mlcoep/sessions/session19\_mlops}
\item Train and save the model, test the service, package it, then check for drift
\item The container step needs Docker; the other steps need only the course environment
\end{itemize}
\begin{lstlisting}[language=bash]
python train_model.py                # creates mpg_model.joblib
python test_api.py                   # good and bad request, no server
uvicorn app:app --port 8000          # real server; stop with Ctrl+C
docker build -t mpg-service .        # package it in a container
docker run -p 8000:8000 mpg-service  # run the container
python drift_check.py                # new data versus training data
\end{lstlisting}
\end{frame}
