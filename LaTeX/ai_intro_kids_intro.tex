\begin{frame}[fragile]\frametitle{What is AI?}
\begin{itemize}
\item AI stands for Artificial Intelligence
\item It's when computers and robots learn to do things that usually need human thinking
\item Like recognizing your face to unlock your phone
\item Or helping you find videos you'll love on YouTube
\item Think of it as teaching machines to be smart!
\end{itemize}
\end{frame}

\begin{frame}[fragile]\frametitle{AI in Your Daily Life}
\begin{itemize}
\item {\bf Video Games:} NPCs (Non-Player Characters) that play against you

\item {\bf Smartphone:} Face unlock, voice assistants like Siri/Google Assistant
