\begin{frame}[fragile]\frametitle{Cost Components of an AI System}
\textbf{Every AI system has these cost buckets, understand them before committing:}
\begin{itemize}
\item \textbf{Model / API Cost:} Pay-per-token (OpenAI, Anthropic, Google) or self-hosted GPU cost
  \begin{itemize}
  \item GPT-4o: $\approx$ \$2.50 / 1M input tokens, \$10 / 1M output tokens
  \item Llama 3 self-hosted: \$0 tokens + GPU cost ($\approx$ \$1--3/hr on A100)
  \end{itemize}
\item \textbf{Compute:} GPU instances for inference (if self-hosted), vector DB, orchestration services
\item \textbf{Storage:} Vector DB storage, prompt/log archives
\item \textbf{Security:} Data encryption, VPC, audit logging, DLP tools
\item \textbf{Tooling:} LangSmith, Langfuse, Promptfoo, dev tooling licenses
\item \textbf{Human Cost:} Prompt engineers, ML engineers, data labelers for evals, governance team
\end{itemize}
\end{frame}

\begin{frame}[fragile]\frametitle{Build vs.\ Buy vs.\ Integrate, Decision Framework}
\begin{center}
\begin{tabular}{|p{2cm}|p{2cm}|p{1cm}| p{1.5cm}|}
\hline
\textbf{Option} & \textbf{When} & \textbf{Cost} & \textbf{Risk} \\
\hline
Use SaaS AI tool & Commodity task & Low & Lock-in \\
Integrate LLM API + RAG & Custom on your data & Medium & Dependency on provider \\
Fine-tune existing LLM & High-volume, specific task & High & Needs labeled data \\
Train own LLM & Core IP, extreme scale & Very High & $>$\$1M budget \\
\hline
\end{tabular}
\end{center}

\textbf{Recommendation for most GTC use cases:} \textbf{Integrate LLM API + RAG}.
\begin{itemize}
\item 90\% of internal use cases solved with API + RAG
\item Fine-tune only when you have $>$10K labeled examples and quality still insufficient
\item Train your own model: almost never justified unless you are a foundation model company
\end{itemize}
\end{frame}

\begin{frame}[fragile]\frametitle{AI ROI, How to Think About It}
\textbf{ROI formula (simplified):}
\[
\text{ROI} = \frac{\text{Value Gained} - \text{Total AI Cost}}{\text{Total AI Cost}} \times 100\%
\]
\textit{This means: for every rupee spent on AI, how many rupees of value do you get back?}

\textbf{Value can be:}
\begin{itemize}
\item \textbf{Time saved:} 40\% reduction in PR review time for 20 engineers = $20 \times 0.4 \times \text{hourly rate}$ per week
\item \textbf{Quality gains:} Bugs caught by AI before production = avoided incidents (quantify per P1 cost)
\item \textbf{Revenue impact:} Faster feature delivery = earlier market capture
\item \textbf{Cost avoidance:} AI handles Tier-1 support = fewer support hires
\end{itemize}

\textbf{Productivity metric frameworks:} SPACE (Satisfaction, Performance, Activity, Communication, Efficiency), DORA metrics with AI
\end{frame}

\begin{frame}[fragile]\frametitle{FinOps for AI, Cost Control Strategies}
\textbf{Token cost is your primary lever, optimize it:}
\begin{itemize}
\item \textbf{Use the right model for the task:} Route simple queries to GPT-4o-mini (\$0.15/1M) not GPT-4o (\$2.50/1M), 16$\times$ cheaper
\item \textbf{Prompt caching:} Cache repeated system prompts; Anthropic offers 90\% discount on cached tokens
\item \textbf{Limit output tokens:} Set \texttt{max\_tokens}, output is 3--4$\times$ more expensive than input
\item \textbf{Batching:} Use Batch API for non-real-time tasks, 50\% cost reduction
\item \textbf{Monitoring dashboards:} Track cost per user, per feature, per team, set budgets and alerts
\item \textbf{Usage governance:} Define who can use which model tier; require justification for expensive models
\end{itemize}

\textbf{Target:} Set cost-per-interaction KPI before launching any AI feature to production.
\end{frame}

\begin{frame}[fragile]\frametitle{Pricing AI Products, Own LLM vs.\ Integrate}
\textbf{If you are \textit{building an AI product} to sell:}
\begin{itemize}
\item \textbf{Integrated LLM model:} Pass-through API cost + margin. Predictable per-call pricing. Fast to market.
  \begin{itemize}
  \item \textit{Risk:} Provider price changes affect your margins. Dependency on third party uptime.
  \end{itemize}
\item \textbf{Own / self-hosted LLM:} Fixed GPU infrastructure cost, variable at scale, better unit economics after breakeven
  \begin{itemize}
  \item \textit{Risk:} Heavy upfront infra investment, MLOps team needed, model maintenance burden
  \end{itemize}
\end{itemize}

\textbf{Pricing models for AI products:}
\begin{itemize}
\item Per-query pricing (e.g., \$X per 1000 API calls)
\item Seat-based (per user/month), bundle AI cost into subscription
\item Usage-based with tiers, free tier limited, paid tier unlimited (freemium)
\item Outcome-based, pay per resolved ticket, per detected bug (hardest to implement)
\end{itemize}
\end{frame}
