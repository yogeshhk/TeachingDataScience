\begin{frame}[fragile]\frametitle{Physical AI, The Next Frontier}
\textbf{AI is leaving the screen and entering the physical world:}
\begin{itemize}
\item \textbf{Robotics:} Boston Dynamics + AI = robots that navigate unstructured environments. NVIDIA Isaac platform for robot training.
\item \textbf{Autonomous Vehicles:} Waymo, Tesla FSD, AI driving at scale
\item \textbf{Smart Manufacturing:} Computer vision for defect detection; robotic arms with LLM-based task planning
\item \textbf{Healthcare:} Robotic surgery (da Vinci AI), AI-guided drug discovery (AlphaFold, Nobel Prize 2024)
\item \textbf{NVIDIA's focus (2025):} ``Physical AI'' as the next major platform shift, combining foundation models with robotics
\end{itemize}

\textbf{For engineering leaders:} Physical AI opens new product opportunities. It also introduces new safety-critical requirements, model correctness is now literally life-or-death.
\end{frame}

\begin{frame}[fragile]\frametitle{AGI and What Lies Ahead}
\textbf{What is AGI?} Artificial General Intelligence, a system that can perform \textit{any intellectual task} a human can, at human-level or better.

\textbf{Timeline predictions:}
\begin{itemize}
\item \textbf{Optimists (2030s):} Sam Altman, Demis Hassabis, we are on a rapid exponential curve
\item \textbf{Skeptics (decades or never):} Yann LeCun, LLMs lack grounding, causality, world models
\end{itemize}

\textbf{What we do know:}
\begin{itemize}
\item Current AI is \textit{narrow but deep}, superhuman at specific tasks, poor at reasoning across domains
\item Key unsolved challenges: common-sense reasoning, causal understanding, reliable long-horizon planning
\end{itemize}

\textbf{Leader takeaway:} Plan for \textit{AI-augmented engineering} in the near term, not AGI. Invest in AI literacy, tooling, and governance now.
\end{frame}

\item QA Engineer $\rightarrow$ AI test strategist (designs eval frameworks, not individual test cases)

\textbf{Skills to develop now:} Prompt engineering, RAG system design, AI eval design, AI governance, cost management for LLMs.
\end{frame}

\begin{frame}[fragile]\frametitle{AI Limits, Know What It Cannot Do}
\begin{itemize}
\item \textbf{No true abstract reasoning:} Can simulate reasoning but lacks genuine causal understanding
\item \textbf{Accuracy limits:} Every model has a hallucination rate, never trust without verification for critical tasks
\item \textbf{Hard to capture tacit knowledge:} Things your team ``just knows'' from years of experience are not in training data
\item \textbf{Interpretability:} Large models are black boxes, explaining \textit{why} a decision was made is still hard
\item \textbf{Context length limits:} Even 1M token models cannot truly ``understand'' a million-token codebase, chunking introduces blind spots
\item \textbf{Real-time awareness:} Without RAG or tools, models are frozen at training cutoff
\end{itemize}
\end{frame}
