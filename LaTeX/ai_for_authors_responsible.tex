% Part B (AI for Authors): responsible use. 4 frames.
% Policy statements are kept general on purpose: authors must read each publisher's own guidelines.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[fragile]\frametitle{Your Manuscript Is Confidential}
\begin{itemize}
\item An unpublished manuscript, a contract, a reviewer's comments and an editor's email are all sensitive.
\item \textbf{Do not paste} them into a public AI tool. Depending on the tool and its settings, what you type may be stored or used to improve the model.
\item Send small pieces, such as one paragraph with names removed, or use a tool whose data terms you have read.
\item For peer review of someone else's paper, do not use AI tools at all unless the journal explicitly allows it.
\end{itemize}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[fragile]\frametitle{Copyright, Publisher Policies and Disclosure}
\begin{itemize}
\item \textbf{Publisher and journal policies differ.} Many ask you to declare any use of AI and some restrict it. Read the guidelines before you submit.
\item \textbf{Disclose what you did}, in plain words: for example, ``AI was used for web search and to prepare diagrams; all text is by the author.''
\item \textbf{Copyright:} do not paste other authors' text into a tool to be reworked, and do not publish AI-made images or text without checking the tool's terms of use.
\item \textbf{Citations:} cite only what you have opened and read.
\end{itemize}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[fragile]\frametitle{Quick Check: Spot the Problem}
\begin{itemize}
\item An author writes this to a public AI chatbot. How many problems can you find?
\end{itemize}
\begin{lstlisting}[language=HTML]
Here is my full unpublished manuscript and the
publisher's draft contract: [pastes both].
Tell me if the contract is fair, and rewrite
Chapter 2 so it sounds more professional.
\end{lstlisting}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[fragile]\frametitle{Quick Check: Answer}
\begin{itemize}
\item \textbf{Confidentiality:} the whole unpublished manuscript and a contract went into a public tool.
\item \textbf{Authorship:} ``rewrite Chapter 2'' puts someone else's words in the text, and may break the publisher's AI policy.
\item \textbf{Legal advice:} a chatbot is not a lawyer. Ask the publisher, a literary agent or an author society about the contract.
\item \textbf{Fix:} ask for an editorial review of one paragraph without names, and keep the rewriting yourself.
\end{itemize}
\end{frame}
