\Large

\medskip
\textbf{AI helps before and after.}
\end{center}

\medskip

\item \textbf{Before:} your own idea and plan first; then research questions and outline stress-tests.

% [ai-for-x: added 2026-11-08 | summary: the spectrum of AI use, from no use to whole books generated]
\begin{frame}[fragile]\frametitle{Summary: The Spectrum of AI Use}
\begin{columns}
    \begin{column}[T]{0.31\linewidth}
    \textbf{No AI at all}
    \begin{itemize}
    \item A valid personal choice
    \item Hard to keep: AI sits inside many tools already
    \end{itemize}
    \end{column}
    \begin{column}[T]{0.34\linewidth}
    \textbf{AI as an assistant}
    \begin{itemize}
    \item Search, fact checks, grammar lists, review, diagrams, emails
    \item You think, plan and write
    \end{itemize}
    \end{column}
    \begin{column}[T]{0.31\linewidth}
    \textbf{Whole books generated}
    \begin{itemize}
    \item Books produced automatically, in bulk
    \item No original thought, and no copyright protection for the text
    \end{itemize}
    \end{column}
\end{columns}
\medskip
\begin{itemize}
\item Ethical use lies in between, close to ``you write''. Use AI as you once used a search engine: as help, not as the author.
\item A test: can you explain and defend every idea and every sentence as your own?
\end{itemize}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{frame}[fragile]\frametitle{Summary: What Stays Yours}
\begin{columns}
    \begin{column}[T]{0.48\linewidth}
    \textbf{Yours alone}
    \begin{itemize}
    \item The idea and your intent
    \item The plan, structure and skeleton
    \item What is new: your USP
    \item Your thought process
    \item The prose, and your judgment
    \end{itemize}
    \end{column}
    \begin{column}[T]{0.48\linewidth}
    \textbf{AI as assistant}
    \begin{itemize}
    \item Web search and verification
    \item Fact checks and grammar lists
    \item Reviews and consistency checks
    \item Diagrams and layout help
    \item Emails and admin
    \end{itemize}
    \end{column}
\end{columns}
\medskip
\begin{center}
\textbf{The thinking, the prose and the novelty are yours.}
\end{center}
\end{frame}

\item Pick \textbf{one} task you dislike or keep putting off: an outline check, a continuity check, a list of search queries or a proposal email.
\item Write one CRAFT prompt for it. Run it, and read the answer critically.
\item Save the prompt that works. That is the start of your own prompt library.
\item Write down how you will describe your AI use in your next submission.
\end{itemize}
