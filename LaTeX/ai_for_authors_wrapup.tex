\begin{frame}[fragile]\frametitle{The One Thing to Remember}
\begin{center}
\Large
\textbf{You write.}

\medskip
\textbf{AI helps before and after.}
\end{center}

\medskip
\begin{itemize}
\item \textbf{Before:} ideas, research questions, outline stress-tests.
\item \textbf{While:} only web searches and diagrams.
\item \textbf{After:} editorial review, layout review, emails.
\end{itemize}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[fragile]\frametitle{Start Small: This Week's Action}
\begin{itemize}
\item Pick \textbf{one} task from your current project: an outline check, a list of search queries or a proposal email.
\item Write one CRAFT prompt for it. Run it, and read the answer critically.
\item Save the prompt that works. That is the start of your own prompt library.
\item Write down how you will describe your AI use in your next submission.
\end{itemize}
\end{frame}
