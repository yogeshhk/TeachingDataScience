\begin{frame}[fragile]\frametitle{When to Use What, Decision Tree}
\begin{itemize}
\item \textbf{Clear rules, deterministic logic?} $\rightarrow$ Traditional programming
\item \textbf{Pattern recognition in structured data?} $\rightarrow$ Classical ML (XGBoost, sklearn)
\item \textbf{Images, speech, complex unstructured data?} $\rightarrow$ Deep Learning / CNN / RNN
\item \textbf{Text generation, Q\&A, summarization, code?} $\rightarrow$ LLM via API
\item \textbf{Q\&A on your private documents?} $\rightarrow$ RAG + LLM
\item \textbf{Multi-step autonomous task on internal systems?} $\rightarrow$ LLM Agent + MCP
\item \textbf{Volume: 10M+ calls/day, very specific task?} $\rightarrow$ Fine-tune SLLM
\item \textbf{Real-time physical-world control?} $\rightarrow$ Physical AI / Robotics
\end{itemize}
\end{frame}

\begin{frame}[fragile]\frametitle{Your 6--12 Month AI Roadmap Template}
\textbf{Agree on these with your leadership team:}
\begin{itemize}
\item \textbf{3--5 Priority Use Cases:} Rank by ROI $\times$ feasibility. Start with quick wins.
\item \textbf{Investment Areas:}
  \begin{itemize}
  \item Tooling (licenses, API access, vector DB)
  \item People (1--2 AI champions per team, 1 MLOps engineer)
  \item Training (this program, prompt engineering workshops)
  \end{itemize}
\item \textbf{Governance Model:} Finalize AI usage policy, approval workflow, audit mechanism
\item \textbf{Milestones:}
  \begin{itemize}
  \item Month 1--2: Internal AI policy published. AI champions identified. First RAG bot live.
  \item Month 3--4: 2--3 team-level AI features in production with evals.
  \item Month 5--6: Internal AI platform (shared tools, shared prompt library). FinOps dashboards live.
  \item Month 7--12: AI in every SDLC phase. Governance mature. ROI measured.
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[fragile]\frametitle{Getting Started, Your Action Plan}

\textbf{For Tech Leads \& Architects:}
\begin{itemize}
\item Build: a RAG bot on your team's runbooks this sprint
\item Evaluate: GitHub Copilot / Cursor for your team (1-month trial)
\item Read: LangChain docs, OpenAI Cookbook
\end{itemize}

\textbf{For Engineering Managers:}
\begin{itemize}
\item Identify: your team's AI champion
\item Draft: team-level AI usage policy (use this deck as input)
\item Measure: baseline DORA metrics before AI adoption to quantify ROI after
\end{itemize}

\textbf{For All Leaders:}
\begin{itemize}
\item Hands-on: Spend 30 min/day with Claude or ChatGPT for real work tasks
\item Stay current: Subscribe to \textit{The Batch} (DeepLearning.AI), \textit{TLDR AI}
\item Think critically: Question every AI claim, ask for the eval results
\end{itemize}

\end{frame}
