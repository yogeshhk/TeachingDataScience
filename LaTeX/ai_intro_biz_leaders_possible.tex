\begin{frame}[fragile]\frametitle{Indian Success Stories in AI}

\begin{itemize}
  \item \textbf{Havells:} AI-driven personalized marketing campaigns, higher conversion, lower CAC.
  \item \textbf{Polycab:} Predictive demand planning reducing stockouts; improved working capital efficiency.
  \item \textbf{Crompton:} AI-powered chatbots for customer service, reduced call center load by 35\%.
  \item \textbf{Bajaj Electricals:} Dealer analytics providing actionable sales insights to field teams.
  \item \textbf{Tata Steel:} Computer vision on production lines, defect detection accuracy improved to 99\%+.
  \item \textbf{HDFC Bank:} AI fraud detection engine processing millions of transactions in real-time.
\end{itemize}

\textbf{Lesson:} These are not moonshots. They are \textit{practical, incremental} AI adoptions with measurable ROI, started with one use case, then scaled.

\end{frame}

\begin{frame}[fragile]\frametitle{Global Inspiration, How Industry Leaders Use AI}

\begin{itemize}
  \item \textbf{Schneider Electric:} Predictive maintenance AI cuts unplanned downtime by 30--40\%.
  \item \textbf{Philips:} AI-personalized customer experience, adaptive product recommendations at scale.
  \item \textbf{LG:} Mining customer feedback with NLP to drive product innovation cycles.
  \item \textbf{Amazon:} AI-powered demand forecasting and warehouse robotics, the backbone of 1-day delivery.
  \item \textbf{Google DeepMind:} AI optimizing Google's own data center cooling, 15\% energy reduction.
  \item \textbf{AlphaFold (DeepMind):} Solved protein folding, accelerated drug discovery by years. Nobel Prize 2024.
\end{itemize}

\textbf{Common success factors across all these:}
\begin{itemize}
  \item Leadership sponsorship and clear business problem
  \item Strong data foundations built \textit{before} AI deployment
  \item Experimentation mindset, start small, measure, scale
\end{itemize}

\end{frame}
