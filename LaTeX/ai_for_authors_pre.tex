\begin{frame}[fragile]\frametitle{Your Own Plan Comes First}
\begin{columns}
    \begin{column}[T]{0.48\linewidth}
    \textbf{Yours alone}
    \begin{itemize}
    \item Why this book: your intent
    \item Who it is for
    \item What is new: your USP
    \item Your structure and skeleton
    \item Your way of thinking
    \end{itemize}
    \end{column}
    \begin{column}[T]{0.48\linewidth}
    \textbf{Then ask AI to test it}
    \begin{itemize}
    \item What is missing or out of order
    \item Which books already cover this
    \item What a sceptical reader would ask
    \end{itemize}
    \end{column}
\end{columns}
\medskip
\begin{itemize}
\item Write your plan on paper first. Do not ask AI to invent the book for you.
\item A book built from AI answers could be anyone's book. Your plan makes it yours.
\end{itemize}
\end{frame}

\begin{frame}[fragile]\frametitle{Before Writing: What AI Is Good For}
\begin{itemize}
\item \textbf{Brainstorming}: more options around your idea fast; you choose and discard.
\item \textbf{Research}: finding what already exists on your topic, and what readers ask about it.
\item \textbf{Structure}: testing an outline for gaps, order and balance.
% [ai-for-x: rewritten 2026-11-08 | AI works on a plan that is already the author's]

\item In each case AI produces candidates for a plan that is already yours. You decide, and you check every fact.
\end{itemize}
\end{frame}

\begin{frame}[fragile]\frametitle{CRAFT Example: Brainstorming Ideas}
\begin{itemize}
\item Ask for many options, then pick. Ask for the opposite view to avoid a one-sided list.
\end{itemize}
\begin{lstlisting}[language=HTML]
You are an experienced non-fiction editor (R).
I am planning a book for first-time managers on
running meetings (C).

Give me 15 possible angles for the book, each in
one line (A). Style: "Meetings as small
experiments: try, measure, adjust" (F).
No jargon; mark the five most unusual (T).
\end{lstlisting}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[fragile]\frametitle{CRAFT Example: Research and Web Search}
\begin{itemize}
\item Use AI to plan the search and summarise what you found. Open the sources yourself.
\item Never cite a source you have not read. AI can invent titles, authors and links.
\end{itemize}
\begin{lstlisting}[language=HTML]
You are a research assistant (R). I am writing on
how remote teams run meetings (C).

List 10 search queries I should try, and for each
the kind of source it should find, such as a study,
a survey or a practitioner blog (A).
Table: Query | Source type | Why it helps (T).
\end{lstlisting}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[fragile]\frametitle{CRAFT Example: Stress-Test Your Outline}
\begin{itemize}
\item Paste your own outline, not text from anyone else's book.
\end{itemize}
\begin{lstlisting}[language=HTML]
You are a sceptical reader who has bought many
books on this topic (R). Here is my chapter outline
for a book on running meetings: [OUTLINE] (C).

Tell me what is missing, what is repeated and
which chapter a reader is likely to skip (A).
Give 5 points, most serious first, 2 lines each (T).
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]\frametitle{CRAFT Example: Facts from Your Own Earlier Book}
\begin{itemize}
\item Use only your own earlier text, for a sequel or a new edition.
\item Check every line against the book. AI can mix up names or invent a detail.
\end{itemize}
\begin{lstlisting}[language=HTML]
You are a careful research assistant (R). Here is
my own earlier book: [EARLIER BOOK TEXT]. I am
writing a sequel and have forgotten the details (C).

List every fact about the people, places and terms
it introduces: names, roles, dates, definitions (A).
Table: Item | Fact | Chapter where it appears (T).
\end{lstlisting}
\end{frame}
