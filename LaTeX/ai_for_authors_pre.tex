% Part B (AI for Authors): before writing. 4 frames.
% Each frame shows one CRAFT prompt. Examples are neutral and invented; no real titles or publishers.

\begin{frame}[fragile]\frametitle{Before Writing: What AI Is Good For}
\begin{itemize}
\item \textbf{Brainstorming}: many ideas fast; you choose and discard.
\item \textbf{Research}: finding what already exists on your topic, and what readers ask about it.
\item \textbf{Structure}: testing an outline for gaps, order and balance.
\item In each case AI produces candidates. You decide, and you check every fact it gives you.
\end{itemize}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[fragile]\frametitle{CRAFT Example: Brainstorming Ideas}
\begin{itemize}
\item Ask for many options, then pick. Ask for the opposite view to avoid a one-sided list.
\end{itemize}
\begin{lstlisting}[language=HTML]
You are an experienced non-fiction editor (R).
I am planning a book for first-time managers on
running meetings (C).

Give me 15 possible angles for the book, each in
one line (A). Style: "Meetings as small
experiments: try, measure, adjust" (F).
No jargon; mark the five most unusual (T).
\end{lstlisting}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[fragile]\frametitle{CRAFT Example: Research and Web Search}
\begin{itemize}
\item Use AI to plan the search and summarise what you found. Open the sources yourself.
\item Never cite a source you have not read. AI can invent titles, authors and links.
\end{itemize}
\begin{lstlisting}[language=HTML]
You are a research assistant (R). I am writing on
how remote teams run meetings (C).

List 10 search queries I should try, and for each
the kind of source it should find, such as a study,
a survey or a practitioner blog (A).
Table: Query | Source type | Why it helps (T).
\end{lstlisting}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[fragile]\frametitle{CRAFT Example: Stress-Test Your Outline}
\begin{itemize}
\item Paste your own outline, not text from anyone else's book.
\end{itemize}
\begin{lstlisting}[language=HTML]
You are a sceptical reader who has bought many
books on this topic (R). Here is my chapter outline
for a book on running meetings: [OUTLINE] (C).

Tell me what is missing, what is repeated and
which chapter a reader is likely to skip (A).
Give 5 points, most serious first, 2 lines each (T).
\end{lstlisting}
\end{frame}
