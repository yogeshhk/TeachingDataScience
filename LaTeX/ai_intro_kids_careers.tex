\begin{frame}[fragile]\frametitle{Jobs When You Grow Up}

{\bf Jobs AI might reduce:}
\begin{itemize}
\item Repetitive tasks (like assembly lines)
\item Simple data entry
\end{itemize}

{\bf NEW jobs AI creates:}
\begin{itemize}
\item AI Trainers (teaching AI to be better)
\item Robot Designers
\item AI Ethics Officers (making sure AI is fair)
\item Prompt Engineers (people who are really good at talking to AI!)
\end{itemize}

{\bf Best skills for the future:}
\begin{itemize}
\item Creativity (AI can't match human imagination!)
\item Critical Thinking
\item Working WITH AI (not against it)
\end{itemize}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[fragile]\frametitle{Careers in AI}
\begin{itemize}
\item {\bf Data Scientist:} Detective who finds patterns in data
\item {\bf Machine Learning Engineer:} Builds AI systems
\item {\bf AI Researcher:} Discovers new AI techniques
\item {\bf Robotics Engineer:} Makes physical robots
\item {\bf Game Developer:} Creates AI for video games
\end{itemize}

\vspace{0.5cm}
All these need: Math, Science, Coding, and Curiosity!
\end{frame}
