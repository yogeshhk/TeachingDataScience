\begin{frame}[fragile]\frametitle{AI Governance, What Must Be in Place}
\textbf{An AI governance framework answers:}
\begin{itemize}
\item \textbf{AI Usage Policy:} What is allowed? What requires review? What is prohibited?
\item \textbf{Approval Workflow:} How does a team get approval to deploy an AI feature to production?
\item \textbf{Review \& Audit:} How are AI outputs reviewed? How long are interactions logged? Who can audit?
\item \textbf{Incident Response:} What happens when AI causes a production issue or generates harmful content?
\item \textbf{Vendor Management:} Which AI providers are approved? What are the contractual data-use clauses?
\item \textbf{Training \& Certification:} Are engineers trained on AI policy before building AI features?
\end{itemize}
\end{frame}

\begin{frame}[fragile]\frametitle{Key AI Risks for Engineering Leaders}
\begin{itemize}
\item \textbf{Hallucinations:} Model confidently states wrong information.
  \textit{Mitigation: RAG grounding, human review for high-stakes outputs, show sources}
\item \textbf{Data Leakage:} Sensitive data sent to external LLM APIs.
  \textit{Mitigation: DLP, approved-only APIs, no PII in prompts policy}
\item \textbf{Bias \& Unfairness:} AI trained on biased data produces biased outputs.
  \textit{Examples: Amazon hiring tool (2018), facial recognition errors for minorities}
\item \textbf{Auditability:} Cannot explain why AI made a decision, regulatory risk.
  \textit{Mitigation: Log all inputs/outputs, use explainable models where regulation demands}
\item \textbf{Regulatory Exposure:}
  \begin{itemize}
  \item EU AI Act (2024): Risk-based regulation; high-risk AI systems require conformity assessment
  \item GDPR: AI decisions using personal data require legal basis + right to explanation
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[fragile]\frametitle{AI Bias, Fairness \& Environmental Cost}
\textbf{Bias Examples:}
\begin{itemize}
\item Amazon hiring AI (2018): downgraded female candidates, trained on male-dominated historical data
\item Facial recognition: higher error rates for people with darker skin tones
\item Loan approval AI: replicating historical redlining patterns
\end{itemize}

\textbf{What's being done:} Fairness metrics, diverse training data, EU AI Act mandatory bias audits.

\vspace{0.5em}
\textbf{Environmental Cost:}
\begin{itemize}
\item Training GPT-3: $\approx$ 552 tonnes $\text{CO}_2$ equivalent
\item Running large LLMs: significant water cooling consumption in data centers
\item Counter-balance: AI optimizing energy grids (DeepMind: 15\% energy reduction), climate modeling
\end{itemize}
\textbf{Message:} ``Innovation with responsibility.''
\end{frame}
