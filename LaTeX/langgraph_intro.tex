\begin{frame}[fragile]\frametitle{The Evolution of LLM Applications}
      \begin{itemize}
        \item \textbf{Early Days}: Simple prompt-response patterns with single LLM calls
        \item \textbf{Challenge}: Real-world problems require multiple steps, decisions, and tool usage
        \item \textbf{Question}: How do we give LLMs more capability while maintaining control?
        \item \textbf{Journey}: From deterministic code → intelligent chains → autonomous agents
        \item \textbf{Core Tension}: Freedom vs Reliability
        \begin{itemize}
            \item More autonomy = More capable, but less predictable
            \item More structure = More reliable, but less flexible
        \end{itemize}
        \item \textbf{Goal}: Find the sweet spot between flexibility and control
      \end{itemize}
\end{frame}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[fragile]\frametitle{Comparison of Autonomy Levels}
\footnotesize
\begin{tabular}{|l|p{3.4cm}|p{4.4cm}|}
\hline
\textbf{Level} & \textbf{Description} & \textbf{Limitation} \\
\hline
\textbf{Code} & 100\% deterministic, rule-based logic. & Must anticipate every scenario. \\
\hline
\textbf{LLM Call} & A single, atomic intelligent task like summarization. & Cannot break down complex, multi-step problems. \\
\hline
\textbf{Chains} & A fixed, unidirectional sequence of operations (A \(\rightarrow\) B \(\rightarrow\) C). & Inflexible; cannot adapt to different inputs or scenarios. \\
\hline
\textbf{Routers} & Chains with conditional branching, but still unidirectional. & No loops or cycles; cannot revisit previous steps to correct mistakes. \\
\hline
\textbf{State (LangGraph)} & A state machine with loops and branches. The workflow is defined by the developer, but the path can be cyclic (A \(\rightarrow\) B \(\leftrightarrow\) C). & The flow is prescribed by the developer; not fully autonomous. \\
\hline
\textbf{Agents} & Fully autonomous decision-making. The agent independently decides the next step in a Think \(\rightarrow\) Act \(\rightarrow\) Observe loop. & Can be unpredictable and is prone to looping infinitely without strict controls. \\
\hline
\end{tabular}
\end{frame}

        \item The evolution moves from simple, deterministic code to complex, autonomous agents capable of cyclic reasoning and self-correction.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[fragile]\frametitle{The Pitch: Controlled Flexibility}
      \begin{itemize}
        \item \textbf{Problem}: Traditional chains are too rigid, pure agents are too unpredictable
        \item \textbf{Solution}: LangGraph provides controlled flexibility through graph-based state machines
        \item \textbf{Key Benefits}:
        \begin{itemize}
            \item Explicit control flow with loops, branches, and conditional logic
            \item Built-in state persistence for long-running workflows
            \item Human-in-the-loop capabilities at any point
            \item Streaming support for real-time feedback
            \item Production-ready with debugging and observability
        \end{itemize}
        \item \textbf{Use Cases}: Multi-step reasoning, workflow automation, complex decision trees, collaborative agents
        \item \textbf{Result}: Reliable autonomous systems that combine flexibility with predictability
      \end{itemize}
\end{frame}

\begin{frame}[fragile]\frametitle{Agents in LangChain vs LangGraph}

\begin{columns}
    \begin{column}[T]{0.6\linewidth}
      \begin{itemize}
        \item \textbf{LangChain Agents (legacy AgentExecutor)}:
        \begin{itemize}
            \item Black-box execution with limited control
            \item Fixed ReAct pattern: a Think, Act, Observe loop until the final answer
            \item Cannot pause, resume, or modify mid-execution
            \item No state persistence across sessions
            \item Prone to infinite loops without guardrails
        \end{itemize}
        \item \textbf{LangGraph Agents}:
        \begin{itemize}
            \item White-box with full visibility into execution
            \item Custom patterns beyond ReAct
            \item Pause, resume, edit state at any node
            \item Persistent state with checkpointing
            \item Explicit cycle limits and control flow
        \end{itemize}
        \item \textbf{Bottom Line}: LangChain agents for simple tasks, LangGraph for production systems
      \end{itemize}

\begin{frame}\frametitle{The Building Blocks}
\begin{itemize}
\item Model: Large Language Model that supports Function Calling
\item Tools: Actions taken by app - API calls, database operations, etc.
\item State: Information carried throughout the workflow (e.g., Message State)
\item Node: Executable logic container - a LangChain runnable or Tool invoker
\item Edge: Control flow of information - conditional or normal
\item Workflow: The graph with nodes and edges that can be invoked or streamed
\end{itemize}

\begin{frame}[fragile]\frametitle{Nodes, Edges and State}


      \begin{itemize}
        \item \textbf{Nodes (N):} Individual processing steps as Python functions that transform state
        \item Each node encapsulates one sub-task: LLM calls, calculations, tool invocations
        \item \textbf{Edges (E):} Directed connections determining execution flow between nodes
        \item Edges can be linear or conditional routes based on current state
        \item \textbf{State (S):} The shared context object, a Pydantic model, that flows through every node
        % \item State enables context and memory across workflow
        % \item StateGraph ties everything together with designated START and END nodes
        % \item Handles interactive, conditional loops that static chains struggle with
      \end{itemize}

\begin{frame}[fragile]\frametitle{State: The Context Object}
\adjustbox{valign=t}{%
\begin{minipage}{0.56\linewidth}
      \begin{itemize}
        \item The \textbf{state} is one context object that flows through every node.
        \item It is a \textbf{Pydantic} class: a Python data model with typed, validated fields.
        \item A node receives the state and returns only the fields it changed.
        \item LangGraph merges the update, then passes the state on.
        \item Edges read the same object to pick the next node.
        \item Not the read-only runtime \texttt{context} of LangGraph 1.0 (for example a user ID), which is passed separately.

\begin{frame}[fragile]\frametitle{Defining and Updating State}
\begin{itemize}
  \item The schema declares every field once, with a type and a default.
  \item A wrong type fails fast with a validation error.
  \item Nodes push partial updates, not full state objects.
  \item \texttt{TypedDict} also works, but Pydantic adds validation and defaults.
\end{itemize}

\begin{lstlisting}[language=Python]
from pydantic import BaseModel, Field

class AgentState(BaseModel):
    messages: list = Field(default_factory=list)
    summary: str = ""
    iteration: int = 0

def my_node(state: AgentState) -> dict:
    # read with attribute access, return only what changed
    return {"iteration": state.iteration + 1}
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]\frametitle{Conditional Edge in Code}
\begin{itemize}
  \item Function evaluates the state to decide the next node.
  \item Mapping links output labels to subsequent graph nodes.
  \item Allows looping or early termination based on conditions.
  \item Useful for iterative reasoning or bounded task execution.
\end{itemize}

\begin{lstlisting}[language=Python]
from langgraph.graph import END

def should_continue(state):
    if state.iteration < 3:
        return "continue"
    return "end"

graph.add_conditional_edges(
    "agent_node",
    should_continue,
    {
        "continue": "action_node",
        "end": END
    }
)
\end{lstlisting}

\begin{frame}[fragile]\frametitle{The Ready-Made MessagesState}
      \begin{itemize}
        \item Chat agents almost always need a \texttt{messages} list, so LangGraph ships \texttt{MessagesState} with the \texttt{add\_messages} reducer built in.
        \item \texttt{add\_messages} appends new messages and merges by message ID, so re-sending a message with the same ID updates it.
        \item Subclass it to add your own fields next to \texttt{messages}.
      \end{itemize}
\begin{lstlisting}[language=Python, basicstyle=\tiny]
from langgraph.graph import MessagesState

class AgentState(MessagesState):
    # "messages" is inherited, with the add_messages reducer
    user_id: str
    session_data: dict
    tool_call_count: int
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]\frametitle{Reducer Functions}
% [upgrade-deck: commented out 2026-10-14 | Task 5c | bullets compressed to short fragments, same facts]
% \item In LangGraph, nodes should return a dictionary containing only the fields they've changed. LangGraph then updates the central state.
% \item By default, a new value for a key overwrites the old one.
% \item To accumulate or modify values instead of overwriting, we use reducer functions.
% \item We can define how state keys should be updated using Python's Annotated type.
      \begin{itemize}
        \item Nodes return only the fields they changed; LangGraph updates the state.
        \item By default, a new value \textbf{overwrites} the old one.
        \item A \textbf{reducer function}, attached with \texttt{Annotated}, says how to combine old and new instead.
      \end{itemize}
\begin{block}{Intuition}
\small Overwrite replaces the page; a reducer adds a new line to it.
\end{block}
\begin{lstlisting}[language=Python, basicstyle=\tiny]
from typing import Annotated
import operator
from langchain_core.messages import AIMessage, AnyMessage
from langgraph.graph.message import add_messages
from pydantic import BaseModel, Field

# Use add_messages to append to the list of messages
# Use operator.add to sum numbers
class AppState(BaseModel):
    messages: Annotated[list[AnyMessage], add_messages] = Field(default_factory=list)
    num_steps: Annotated[int, operator.add] = 0

# A node can now return just the new message and a number to add
def my_node(state: AppState) -> dict:
    return {"messages": [AIMessage(content="Hello!")], "num_steps": 1}
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]\frametitle{Quick Check: Core Concepts}
\begin{block}{Think About It}
Node A returns \texttt{\{"notes": ["a"]\}}, then Node B returns \texttt{\{"notes": ["b"]\}}. The \texttt{notes} field has \textbf{no} reducer. What does \texttt{state.notes} hold at the end?
\begin{enumerate}
  \item[A.] \texttt{["a", "b"]}
  \item[B.] \texttt{["b"]}
  \item[C.] \texttt{["a"]}, because the first write wins
  \item[D.] An error, because two nodes cannot write the same key
\end{enumerate}
\end{block}
\end{frame}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[fragile]\frametitle{Quick Check: Core Concepts}
\begin{block}{Answer}
\textbf{Correct: B.} Without a reducer, each update overwrites the field, so the last write (Node B) wins. Option A is what you get with a list reducer such as \texttt{operator.add}; nothing here raises an error.
\end{block}
\end{frame}

\begin{frame}[fragile]\frametitle{Define the State}
      \begin{itemize}
        \item The state is the context object every node receives and updates.
        \item It is a Pydantic \texttt{BaseModel}: typed fields with defaults.
        \item \texttt{add\_messages} appends to \texttt{messages} instead of overwriting it.
      \end{itemize}
\begin{lstlisting}[language=Python, basicstyle=\tiny]
from typing import Annotated
from langchain_core.messages import AIMessage, AnyMessage
from langgraph.graph.message import add_messages
from pydantic import BaseModel, Field

class EmailState(BaseModel):
    email_content: str
    is_spam: bool = False
    draft_response: str = ""
    # Each time a node returns a "messages" key, it is appended to this list
    messages: Annotated[list[AnyMessage], add_messages] = Field(default_factory=list)
\end{lstlisting}
\end{frame}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[fragile]\frametitle{Node Functions: Read and Classify}
      \begin{lstlisting}[language=Python, basicstyle=\tiny]
import os
from langchain_groq import ChatGroq

llm = ChatGroq(model="openai/gpt-oss-20b")

# This node doesn't modify the state, so it can just return an empty dictionary
def read_email(state: EmailState) -> dict:
    return {}

def classify_email(state: EmailState) -> dict:
    # The state arrives as an EmailState object: use attribute access
    prompt = f"Classify this email as spam or legitimate:\n{state.email_content}\nRespond with only 'spam' or 'legitimate'."
    response = llm.invoke(prompt)
    # startswith, because "not spam" also contains the word "spam"
    is_spam = response.content.strip().lower().startswith("spam")
    # Return ONLY the fields that have changed
    return {
        "is_spam": is_spam,
        "messages": [AIMessage(content=response.content, name="classifier")]
    }
      \end{lstlisting}
\end{frame}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[fragile]\frametitle{Node Functions: Handle Spam and Draft}
      \begin{lstlisting}[language=Python, basicstyle=\tiny]
def handle_spam(state: EmailState) -> dict:
    # Return ONLY the new message to be added to the state
    return {
        "messages": [AIMessage(content="Email marked as spam and moved to junk.", name="system")]
    }

def draft_response(state: EmailState) -> dict:
    prompt = f"Draft a professional response to:\n{state.email_content}"
    response = llm.invoke(prompt)
    # Return ONLY the changed fields
    return {
        "draft_response": response.content,
        "messages": [AIMessage(content=response.content, name="drafter")]
    }
      \end{lstlisting}
\end{frame}

\begin{frame}[fragile]\frametitle{Build the Graph with Conditional Routing}
      \begin{itemize}
        \item \texttt{StateGraph(EmailState)} declares the context object the whole graph shares.
        \item The routing function reads the state; the \textbf{path map} turns its output into a node name.
      \end{itemize}
\begin{lstlisting}[language=Python, basicstyle=\tiny]
from langgraph.graph import StateGraph, START, END

workflow = StateGraph(EmailState)
workflow.add_node("read_email", read_email)
workflow.add_node("classify_email", classify_email)
workflow.add_node("spam_handler", handle_spam)
workflow.add_node("response_drafter", draft_response)
workflow.add_edge(START, "read_email")
workflow.add_edge("read_email", "classify_email")

def route_email(state: EmailState) -> str:
    return "spam" if state.is_spam else "not_spam"

# path map: routing output -> node name
workflow.add_conditional_edges(
    "classify_email", route_email,
    {"spam": "spam_handler", "not_spam": "response_drafter"})
workflow.add_edge("spam_handler", END)
workflow.add_edge("response_drafter", END)
app = workflow.compile()
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]\frametitle{Run the Graph}

      \begin{itemize}
        \item Build the initial state; the graph passes it through the nodes and routes on it.
        \item \texttt{invoke} returns the final state as a dict; fields no node touched are absent, so use \texttt{.get}.

\begin{frame}[fragile]\frametitle{Human-in-the-Loop: interrupt() and Command}
      \begin{itemize}
        \item \texttt{interrupt()} pauses inside a node and hands a payload (question, draft) to the human.
        \item The graph needs a checkpointer; the run resumes on the same \texttt{thread\_id}.
        \item \texttt{Command(resume=...)} supplies the human's answer, which becomes the return value of \texttt{interrupt()}.
      \end{itemize}
{\tiny (Ref: Building AI Agents with LangGraph: A Practical Guide - DEV Community)}

\begin{lstlisting}[language=Python, basicstyle=\tiny]
from langgraph.types import interrupt, Command
from langchain_core.messages import AIMessage

def sensitive_action(state: EmailState):
    decision = interrupt({"question": "Approve sending this email?",
                          "draft": state.draft_response})
    if decision["approved"]:
        return {"messages": [AIMessage(content="Email sent!")]}
    return {"messages": [AIMessage(content="Email cancelled.")]}

# Later, resume with the human's decision
app.invoke(Command(resume={"approved": True}), config=config)
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]\frametitle{State Persistence and Checkpointing}
      \begin{itemize}
        \item \textbf{What}: Automatically save state at each node execution
        \item \textbf{Benefits}:
        \begin{itemize}
            \item Resume after failures or interruptions
            \item Time-travel debugging (replay from any checkpoint)
            \item Support long-running workflows across sessions
            \item Enable human-in-the-loop patterns
        \end{itemize}
        \item \textbf{Checkpoint Storage Options}:
        \begin{itemize}
            \item In-memory (development/testing)
            \item SQLite (local persistence)
            \item PostgreSQL (production)
            \item Redis (distributed systems)
        \end{itemize}
        \item \textbf{Key Feature}: Each checkpoint is immutable and versioned
        \item Can fork from any checkpoint to explore alternative paths
      \end{itemize}
