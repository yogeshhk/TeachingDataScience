\begin{frame}[fragile]\frametitle{How to Drive AI Adoption, The Leadership Playbook}
\textbf{What kills AI adoption:}
\begin{itemize}
\item Random experimentation, 20 teams, 20 tools, no shared learning
\item Fear and distrust, engineers worried AI will replace them
\item No clear ownership, everyone waits, no one acts
\end{itemize}

\textbf{What works:}
\begin{itemize}
\item \textbf{AI Champions:} Identify 1--2 enthusiastic engineers per team. Give them time, budget, and air cover to experiment and teach.
\item \textbf{Internal AI Platform:} Centralized, pre-approved AI tools, shared prompt library, shared embedding pipeline. Reduces redundant work, ensures compliance.
\item \textbf{Show wins fast:} Pick one high-value, low-risk use case per team. Show a working demo within 3 weeks. Builds confidence.
\item \textbf{Normalize AI in reviews:} Include AI-assisted PRs in code reviews. Normalize AI-generated first-draft docs.
\end{itemize}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[fragile]\frametitle{What Leadership Must Change}
\textbf{Estimation and planning:}
\begin{itemize}
\item Sprint velocity estimates need to account for AI-assisted development (adjust story points)
\item Some tasks that took 2 days now take 2 hours, your roadmap timelines change
\end{itemize}

\textbf{Review processes:}
\begin{itemize}
\item AI-generated code still needs review, but the focus shifts from typos to logic and architecture
\item Add AI output review as an explicit step in Definition of Done
\end{itemize}

\textbf{Quality gates:}
\begin{itemize}
\item Require eval scores above threshold before AI feature goes to production
\item Mandatory human-in-loop for customer-facing AI outputs in first 3 months
\end{itemize}

\textbf{Expectations:}
\begin{itemize}
\item AI is a \textit{force multiplier}, not a replacement. Outcomes improve; team size stays (or grows for higher-value work).
\end{itemize}
\end{frame}
