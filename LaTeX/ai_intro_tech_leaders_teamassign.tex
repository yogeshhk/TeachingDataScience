\begin{frame}[fragile]\frametitle{Team Assignment, Structure and Deliverables}
\textbf{Objective:} Each group builds a practical AI solution that improves one SDLC phase.

\textbf{Example projects:}
\begin{itemize}
\item PR review agent | Test case generation agent | Support ticket triage agent
\item RCA agent | Requirements $\rightarrow$ design $\rightarrow$ code agent | Data quality monitoring agent
\end{itemize}

\textbf{Deliverables per team:}
\begin{itemize}
\item \textbf{Working demo} (live or recorded)
\item \textbf{Repo with:} code, prompt design (versioned), architecture diagram, cost estimate
\item \textbf{10-min presentation:} Business impact, practicality, reusability
\end{itemize}

\textbf{Format:} Team picks use case + leader. Optional async weekly check-in. Lightweight, respect leadership schedules.
\end{frame}
