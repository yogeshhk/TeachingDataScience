\begin{frame}[fragile]\frametitle{How YOU Learn New Things}
\begin{itemize}
\item Remember when you learned to ride a bicycle?
\begin{itemize}
\item First time: You fell down (made mistakes)
\item Practiced more: Got better each time
\item After many tries: You can ride without thinking!
\end{itemize}
\item Same with learning math, cricket, or dancing
\item Practice + Mistakes = Learning!
\end{itemize}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[fragile]\frametitle{How Computers Learn (Machine Learning)}
\begin{itemize}
\item Computers learn the SAME WAY!
\item Example: Teaching computer to recognize cats
\begin{itemize}
\item Show 1000 pictures of cats
\item Show 1000 pictures of dogs
\item Computer finds patterns (whiskers, pointy ears, etc.)
\item Now it can tell cats from dogs in NEW pictures!
\end{itemize}
\item This is called {\bf Machine Learning} (ML)
\end{itemize}
\end{frame}

\begin{frame}[fragile]\frametitle{Example 1: Ice Cream Sales}
\begin{itemize}
\item {\bf Simple Pattern:} When it's hot, more people buy ice cream
\item {\bf Relationship:} Temperature $\uparrow$ = Ice cream sales $\uparrow$
\item This is EASY to program with simple rules
\item No AI needed!
\end{itemize}

\vspace{0.5cm}
{\bf Rule:} IF temperature $>$ 30°C THEN more ice cream will sell
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[fragile]\frametitle{Example 2: Video Game Player}
\begin{itemize}
\item {\bf Complex Pattern:} What's the best move in a chess game?
\item Too many possibilities to write simple rules!
\item Need to learn from MILLIONS of games
\item AI can do this!
\end{itemize}

\vspace{0.5cm}
How? By playing games over and over, learning what works and what doesn't!
\end{frame}

\begin{frame}[fragile]\frametitle{The AI Family Tree}
\begin{itemize}
\item {\bf Artificial Intelligence (AI):} The big family
\begin{itemize}
\item Making computers smart
\end{itemize}
\item {\bf Machine Learning (ML):} A special member
\begin{itemize}
\item Learning from examples (like the cat/dog pictures)
\end{itemize}
\item {\bf Deep Learning (DL):} The super smart cousin
\begin{itemize}
\item Uses ``artificial brain'' with layers (like your brain has neurons!)
\end{itemize}
\item {\bf Generative AI:} The creative one
\begin{itemize}
\item Can create NEW things: pictures, stories, music!
\item Example: ChatGPT, DALL-E
\end{itemize}
\end{itemize}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[fragile]\frametitle{Three Ways Computers Learn}

{\bf 1. Supervised Learning} (Learning with a teacher)
\begin{itemize}
\item Like learning with answer key
\item Example: Showing labeled pictures - ``This is a cat'', ``This is a dog''
\end{itemize}

{\bf 2. Unsupervised Learning} (Learning by yourself)
\begin{itemize}
\item Finding patterns without help
\item Example: Grouping similar songs together on Spotify
\end{itemize}

{\bf 3. Reinforcement Learning} (Learning by trial and error)
\begin{itemize}
\item Learn by rewards and punishments
\item Example: Game AI that gets points for winning
\end{itemize}

\end{frame}
