% Helper macros for the promo_* slides (books, community). Safe to \input more than once.
%
% \promocover{file.jpg}{label}  : book cover. Shows images/file.jpg if that file exists, otherwise a framed
%                                  placeholder reading <label>.
% \promoqr{file.png}{label}     : QR code with its name underneath. Shows images/file.png if that file exists,
%                                  otherwise an empty frame of the same size, so the name stays and the QR can be
%                                  added later.
%
% To fill a placeholder, drop the real file into images/ under the name used in the promo file and rebuild; no .tex
% edit is needed. Paths are relative to the LaTeX folder (decks and cheat sheets are compiled there).
\providecommand{\promocover}[2]{%
  \IfFileExists{images/#1}{\includegraphics[width=0.8\linewidth,keepaspectratio]{#1}}%
  {\fbox{\parbox[c][0.45\textheight][c]{0.7\linewidth}{\centering\small Cover image\\[0.3em] #2\\[0.3em] (to be added)}}}%
}
\providecommand{\promoqr}[2]{%
  \begin{minipage}[t]{0.45\linewidth}\centering
  \IfFileExists{images/#1}{\includegraphics[width=\linewidth,keepaspectratio]{#1}}%
  {\fbox{\rule{0pt}{0.2\textheight}\hspace{0.2\textheight}}}\\[0.2em]
  {\scriptsize #2}%
  \end{minipage}%
}
