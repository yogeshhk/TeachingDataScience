\begin{frame}[fragile]\frametitle{Enterprise AI Tooling Stack}
\begin{itemize}
\item \textbf{Model Access Layer:}
  Azure OpenAI / AWS Bedrock / Google Vertex AI, enterprise-grade APIs with compliance, rate limits, VPC
\item \textbf{Orchestration:}
  LangChain, LlamaIndex, LangGraph, chain LLM calls, manage memory, build agent workflows
\item \textbf{Vector Databases:}
  Pinecone, Weaviate, pgvector, Chroma, store embeddings for semantic search / RAG
\item \textbf{Observability:}
  LangSmith, Langfuse, Phoenix, trace every LLM call, measure quality, debug failures
\item \textbf{Deployment:}
  Docker + Kubernetes for containerized AI services; model serving via vLLM / Ollama for open models
\item \textbf{IDE Integration:}
  GitHub Copilot, Cursor, Codeium, developer productivity layer
\end{itemize}
\end{frame}

\begin{frame}[fragile]\frametitle{AI Usage Policy, What Must Be Defined}
\textbf{Your org needs clear answers to:}
\begin{itemize}
\item \textbf{Data Classification:} What data can go to external LLM APIs? (PII, IP, secrets, never!)
\item \textbf{Approved Tools:} Which AI tools are pre-approved? Which require a security review?
\item \textbf{Output Ownership:} Who owns AI-generated code? How is it reviewed before merge?
\item \textbf{Audit Trail:} Are LLM interactions logged? Can they be replayed for compliance?
\item \textbf{Model Selection:} Is there a preferred model per use case? Who approves new models?
\item \textbf{Access Control:} Who can deploy AI services? Who approves production use of agents?
\end{itemize}

\textbf{Suggested starting point:} An internal ``AI Use Policy'' document, reviewed by Security + Legal + Engineering leads. Treat AI like you treat external SaaS.
\end{frame}

\begin{frame}[fragile]\frametitle{Key Security Risks in Enterprise AI}
\begin{itemize}
\item \textbf{Data Leakage:} Developer pastes internal credentials or PII into ChatGPT, it may be used for training
  \begin{itemize}
  \item \textit{Mitigation:} Use enterprise API (data not used for training), DLP tools on clipboard
  \end{itemize}
\item \textbf{Prompt Injection:} Malicious input tricks the AI agent into performing unauthorized actions
  \begin{itemize}
  \item \textit{Mitigation:} Input sanitization, tool-call allowlists, human approval for destructive actions
  \end{itemize}
\item \textbf{Hallucinations in Production:} Model confidently returns wrong API schema, wrong config
  \begin{itemize}
  \item \textit{Mitigation:} Retrieval-Augmented Generation (RAG) grounding, human review for critical decisions
  \end{itemize}
\item \textbf{Supply Chain Risk:} Malicious or poisoned open-source model weights
  \begin{itemize}
  \item \textit{Mitigation:} Only use models from verified registries (Hugging Face verified, official providers)
  \end{itemize}
\end{itemize}
\end{frame}
