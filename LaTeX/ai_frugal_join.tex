\section[Join]{Join In}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[fragile]\frametitle{Join In}
\begin{itemize}
\item \textbf{FrugalAI repository}: measured benchmarks and recipes for small models on old laptops and Raspberry Pi, with a small harness to measure your own device.
\item \textbf{Awesome list}: add your frugal AI effort. Free to use, a downloadable model, no per-token charge, hardware stated.
\item \textbf{Measure}: add one result from a device you own. That is the easiest first contribution.
\item \textbf{Students and educators}: projects and internships are open. Start by porting one recipe to your device and measuring it.
\end{itemize}
\end{frame}
